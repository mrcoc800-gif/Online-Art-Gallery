        // Live Bidding System Logic
        let auctionBids = {

        function placeBid(lotId) {
            const input = document.getElementById(`bidInput${lotId}`);
            const newBid = parseFloat(input.value);
            const currentBid = auctionBids[lotId];

            if (isNaN(newBid) || newBid <= currentBid) {
                showToast(`Bid must be higher than $${currentBid.toLocaleString()}`);
                return;
            }

            auctionBids[lotId] = newBid;
            document.getElementById(`bidPrice${lotId}`).innerText = `$${newBid.toLocaleString()}`;
            document.getElementById(`bidder${lotId}`).innerText = "Updated by You";
            input.value = '';
            showToast(`Bid of $${newBid.toLocaleString()} successfully recorded!`);
        }
